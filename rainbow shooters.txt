<script>
"use strict";

window.addEventListener("load", function () {
  const canvas = document.getElementById("game");
  const ctx = canvas.getContext("2d");

  if (!ctx) {
    document.getElementById("message").textContent =
      "Sorry! Your browser could not start the game. Try Chrome or Edge.";
    return;
  }

  const W = canvas.width, H = canvas.height;
  const $ = id => document.getElementById(id);

  let level = 1, player, bot, walls, bullets, sparkles;
  let mouse = {x:850,y:280};
  let firing = false, lastShot = 0, frame = 0;
  let won = false;
  let musicStarted = false;
  let audio = null;

  const wallSets = {
    1: [
      [120,130,145,20,"#ff55ad"],
      [300,220,22,110,"#37dcff"],
      [410,105,140,20,"#ffcf4c"],
      [465,365,160,22,"#c875ff"],
      [665,170,22,115,"#59f58d"],
      [790,420,140,22,"#ff8353"],
      [140,420,150,22,"#45aaff"],
      [885,300,22,85,"#ff63c9"]
    ],
    2: [
      [100,145,155,22,"#ff55ad"],
      [310,245,22,115,"#ffcf4c"],
      [415,115,145,22,"#59f58d"],
      [470,380,165,22,"#c875ff"],
      [665,185,22,115,"#45aaff"],
      [790,430,145,22,"#ff8353"],
      [145,430,155,22,"#b875ff"],
      [880,300,22,90,"#59f58d"]
    ],
    3: [
      [115,155,145,22,"#b875ff"],
      [300,225,22,120,"#45eaff"],
      [420,105,145,22,"#ff63c9"],
      [470,390,165,22,"#ffcf4c"],
      [675,180,22,120,"#59f58d"],
      [790,425,145,22,"#ff8353"],
      [145,430,155,22,"#7187ff"],
      [880,300,22,90,"#ff63c9"]
    ]
  };

  function reset() {
    level = Number($("level").value);
    $("levellabel").textContent = level;
    player = {x:55,y:280,r:15,hp:100,color:"#21cfff"};
    bot = {x:945,y:280,r:17,hp:100,color:"#ff4d91"};
    walls = wallSets[level].map(a => ({
      x:a[0],y:a[1],w:a[2],h:a[3],c:a[4]
    }));
    bullets = [];
    sparkles = [];
    won = false;
    firing = false;
    $("message").textContent =
      "Aim at the bot and fire your flower power! 🌷";
    hud();
  }

  function hud() {
    $("php").textContent = player.hp;
    $("bhp").textContent = bot.hp;
    $("pbar").style.width = player.hp + "%";
    $("bbar").style.width = bot.hp + "%";
  }

  function rounded(x,y,w,h,r,color) {
    ctx.fillStyle = color;
    ctx.beginPath();
    ctx.moveTo(x+r,y);
    ctx.arcTo(x+w,y,x+w,y+h,r);
    ctx.arcTo(x+w,y+h,x,y+h,r);
    ctx.arcTo(x,y+h,x,y,r);
    ctx.arcTo(x,y,x+w,y,r);
    ctx.closePath();
    ctx.fill();
  }

  function background() {
    const hue = (frame*.45)%360;
    const g = ctx.createLinearGradient(
      Math.sin(frame*.008)*300,0,
      W+Math.cos(frame*.007)*250,H
    );
    g.addColorStop(0,`hsl(${hue},95%,63%)`);
    g.addColorStop(.3,`hsl(${hue+70},95%,60%)`);
    g.addColorStop(.6,`hsl(${hue+160},95%,62%)`);
    g.addColorStop(1,`hsl(${hue+260},95%,65%)`);
    ctx.fillStyle = g;
    ctx.fillRect(0,0,W,H);

    // Moving rainbow arches.
    for(let k=0;k<3;k++) {
      const x = ((frame*(.35+k*.13)+k*400)%(W+400))-200;
      const y = 100+k*175+Math.sin(frame*.012+k)*30;
      const colors = [
        "#ff4d9c","#ffad38","#fff16c","#57f59a",
        "#43dcff","#7185ff","#d475ff"
      ];
      ctx.save();
      ctx.globalAlpha=.35;
      ctx.lineWidth=13;
      colors.forEach((c,i)=>{
        ctx.strokeStyle=c;
        ctx.beginPath();
        ctx.arc(x,y,55+i*11,Math.PI,Math.PI*2);
        ctx.stroke();
      });
      ctx.restore();
    }

    // Twinkling stars.
    for(let i=0;i<45;i++) {
      const x=(i*137+frame*(.3+i%3*.15))%W;
      const y=(i*83+Math.sin(frame*.025+i)*22)%H;
      ctx.globalAlpha=.4+.5*Math.abs(Math.sin(frame*.04+i));
      ctx.fillStyle=["#fff","#fff38b","#ffd5f8","#c9ffff"][i%4];
      ctx.beginPath();
      ctx.arc(x,y,2+i%3,0,Math.PI*2);
      ctx.fill();
    }
    ctx.globalAlpha=1;

    ctx.strokeStyle="white";
    ctx.lineWidth=5;
    ctx.strokeRect(4,4,W-8,H-8);
  }

  function drawWalls() {
    walls.forEach(w=>{
      ctx.save();
      ctx.shadowColor=w.c;
      ctx.shadowBlur=18;
      rounded(w.x,w.y,w.w,w.h,7,w.c);
      ctx.shadowBlur=0;
      ctx.strokeStyle="white";
      ctx.lineWidth=2;
      ctx.strokeRect(w.x+1,w.y+1,w.w-2,w.h-2);
      ctx.restore();
    });
  }

  function flower(x,y,size,rotation,hue) {
    ctx.save();
    ctx.translate(x,y);
    ctx.rotate(rotation);
    ctx.shadowColor=`hsl(${hue},100%,70%)`;
    ctx.shadowBlur=12;

    // Five flower petals.
    for(let i=0;i<5;i++) {
      ctx.rotate(Math.PI*2/5);
      ctx.fillStyle=`hsl(${(hue+i*18)%360},100%,70%)`;
      ctx.beginPath();
      ctx.ellipse(0,-size*.62,size*.36,size*.57,0,0,Math.PI*2);
      ctx.fill();
      ctx.strokeStyle="rgba(255,255,255,.8)";
      ctx.lineWidth=1;
      ctx.stroke();
    }

    ctx.shadowBlur=0;
    ctx.fillStyle="#ffeb73";
    ctx.beginPath();
    ctx.arc(0,0,size*.34,0,Math.PI*2);
    ctx.fill();
    ctx.fillStyle="#fff";
    ctx.beginPath();
    ctx.arc(-size*.09,-size*.07,size*.07,0,Math.PI*2);
    ctx.arc(size*.09,-size*.07,size*.07,0,Math.PI*2);
    ctx.fill();
    ctx.restore();
  }

  function character(p,angle) {
    ctx.save();
    ctx.translate(p.x,p.y);
    ctx.rotate(angle);
    ctx.shadowColor=p.color;
    ctx.shadowBlur=12;
    rounded(3,-5,30,10,4,"white");
    rounded(7,-3,26,6,3,p.color);
    ctx.restore();

    ctx.save();
    ctx.translate(p.x,p.y);
    ctx.shadowColor=p.color;
    ctx.shadowBlur=13;
    ctx.fillStyle=p.color;
    ctx.beginPath();
    ctx.arc(0,-7,12,0,Math.PI*2);
    ctx.fill();
    ctx.fillRect(-8,3,16,16);
    ctx.fillRect(-11,15,7,11);
    ctx.fillRect(4,15,7,11);
    ctx.shadowBlur=0;
    ctx.fillStyle="white";
    ctx.beginPath();
    ctx.arc(-4,-9,2,0,Math.PI*2);
    ctx.arc(4,-9,2,0,Math.PI*2);
    ctx.fill();
    ctx.restore();

    rounded(p.x-22,p.y-36,44,6,3,"#45366e");
    rounded(p.x-21,p.y-35,42*p.hp/100,4,2,
      p.hp>30?"#5aff9c":"#ff4545");
  }

  function shoot() {
    if(won || player.hp<=0) return;
    const now=performance.now();
    if(now-lastShot<170) return;
    lastShot=now;

    const a=Math.atan2(mouse.y-player.y,mouse.x-player.x);
    const speed=470;
    bullets.push({
      x:player.x+Math.cos(a)*28,
      y:player.y+Math.sin(a)*28,
      vx:Math.cos(a)*speed,
      vy:Math.sin(a)*speed,
      r:9,life:14,hue:Math.random()*360,spin:0
    });
    for(let i=0;i<4;i++) {
      sparkles.push({
        x:player.x+Math.cos(a)*25,
        y:player.y+Math.sin(a)*25,
        vx:(Math.random()-.5)*70,
        vy:(Math.random()-.5)*70,
        life:.35,hue:Math.random()*360
      });
    }
  }

  function hitRect(x,y,r,w) {
    const cx=Math.max(w.x,Math.min(x,w.x+w.w));
    const cy=Math.max(w.y,Math.min(y,w.y+w.h));
    return (x-cx)**2+(y-cy)**2<r*r;
  }

  function burst(x,y) {
    for(let i=0;i<24;i++) {
      const a=Math.random()*Math.PI*2;
      const speed=40+Math.random()*160;
      sparkles.push({
        x,y,vx:Math.cos(a)*speed,vy:Math.sin(a)*speed,
        life:.7,hue:Math.random()*360
      });
    }
  }

  function update(dt) {
    frame++;
    if(firing) shoot();

    for(let i=bullets.length-1;i>=0;i--) {
      const b=bullets[i];
      const ox=b.x, oy=b.y;
      b.x+=b.vx*dt;
      b.y+=b.vy*dt;
      b.life-=dt;
      b.spin+=dt*5;

      // Bounce off the arena boundaries.
      if(b.x-b.r<10) { b.x=10+b.r; b.vx=Math.abs(b.vx); }
      if(b.x+b.r>W-10) { b.x=W-10-b.r; b.vx=-Math.abs(b.vx); }
      if(b.y-b.r<10) { b.y=10+b.r; b.vy=Math.abs(b.vy); }
      if(b.y+b.r>H-10) { b.y=H-10-b.r; b.vy=-Math.abs(b.vy); }

      // Reflect off walls; never remove a bullet because of a wall.
      for(const w of walls) {
        if(!hitRect(b.x,b.y,b.r,w)) continue;

        const overlapX = Math.min(b.x+b.r-w.x,w.x+w.w-(b.x-b.r));
        const overlapY = Math.min(b.y+b.r-w.y,w.y+w.h-(b.y-b.r));

        if(overlapX<overlapY) {
          b.vx*=-1;
          b.x=ox;
        } else {
          b.vy*=-1;
          b.y=oy;
        }

        b.hue=(b.hue+70)%360;
        break;
      }

      // The enemy can be hit.
      if(Math.hypot(b.x-bot.x,b.y-bot.y)<bot.r+b.r) {
        bot.hp=Math.max(0,bot.hp-25);
        burst(b.x,b.y);
        bullets.splice(i,1);
        hud();

        if(bot.hp<=0) {
          won=true;
          burst(bot.x,bot.y);
          $("message").textContent=
            "🌸🌈 YOU WIN! Press ENTER for the next level! 🎉";
        }
        continue;
      }

      // Bullets eventually expire so the game stays smooth.
      if(b.life<=0) bullets.splice(i,1);
    }

    for(let i=sparkles.length-1;i>=0;i--) {
      const p=sparkles[i];
      p.x+=p.vx*dt;
      p.y+=p.vy*dt;
      p.vx*=.98;
      p.vy*=.98;
      p.life-=dt;
      if(p.life<=0) sparkles.splice(i,1);
    }
  }

  function draw() {
    background();
    drawWalls();

    const pa=Math.atan2(mouse.y-player.y,mouse.x-player.x);
    const ba=Math.atan2(player.y-bot.y,player.x-bot.x);
    character(player,pa);
    character(bot,ba);

    bullets.forEach(b=>flower(b.x,b.y,9,b.spin,b.hue));

    sparkles.forEach(p=>{
      ctx.globalAlpha=Math.min(1,p.life*2);
      flower(p.x,p.y,3,frame*.03,p.hue);
    });
    ctx.globalAlpha=1;

    if(won) {
      ctx.fillStyle="rgba(35,15,90,.78)";
      ctx.fillRect(230,215,540,120);
      ctx.textAlign="center";
      ctx.fillStyle="white";
      ctx.font="bold 36px system-ui";
      ctx.fillText("🌸 YOU WIN! 🌈",W/2,263);
      ctx.font="bold 18px system-ui";
      ctx.fillText("Press ENTER for the next level",W/2,300);
    }
  }

  let last=performance.now();
  function loop(now) {
    const dt=Math.min((now-last)/1000,.035);
    last=now;
    update(dt);
    draw();
    requestAnimationFrame(loop);
  }

  function pointer(e) {
    const r=canvas.getBoundingClientRect();
    mouse.x=(e.clientX-r.left)*W/r.width;
    mouse.y=(e.clientY-r.top)*H/r.height;
  }

  canvas.addEventListener("pointermove",pointer);
  canvas.addEventListener("pointerdown",e=>{
    pointer(e);
    startMusic();
    firing=true;
    shoot();
  });
  window.addEventListener("pointerup",()=>firing=false);
  window.addEventListener("pointercancel",()=>firing=false);

  $("fire").addEventListener("pointerdown",e=>{
    e.preventDefault();
    startMusic();
    firing=true;
    shoot();
  });
  $("fire").addEventListener("pointerup",()=>firing=false);
  $("fire").addEventListener("pointerleave",()=>firing=false);
  $("fire").addEventListener("pointercancel",()=>firing=false);

  $("restart").addEventListener("click",()=>{
    startMusic();
    reset();
  });
  $("level").addEventListener("change",reset);

  document.addEventListener("keydown",e=>{
    startMusic();
    if(e.code==="Space") {
      e.preventDefault();
      firing=true;
      shoot();
    }
    if(e.key.toLowerCase()==="r") reset();
    if(e.key==="Enter" && won) {
      $("level").value=String(level%3+1);
      reset();
    }
  });
  document.addEventListener("keyup",e=>{
    if(e.code==="Space") firing=false;
  });
  window.addEventListener("blur",()=>firing=false);

  // Original flower-and-rainbow music, made with browser audio.
  // Browsers require a click/key press before audio can start.
  function startMusic() {
    if(musicStarted) return;
    const AudioContextClass=window.AudioContext||window.webkitAudioContext;
    if(!AudioContextClass) {
      $("message").textContent=
        "The game works, but this browser does not support its music.";
      return;
    }

    try {
      audio=new AudioContextClass();
      audio.resume();
      musicStarted=true;

      // A cheerful original melody with twinkly high notes.
      const melody=[
        523.25,659.25,783.99,659.25,
        587.33,698.46,880,698.46,
        523.25,659.25,783.99,1046.5,
        880,783.99,659.25,587.33,
        659.25,783.99,880,783.99,
        698.46,880,1046.5,880
      ];
      const bass=[130.81,164.81,196,174.61,146.83,174.61,220,196];
      let step=0;

      function note(freq,start,duration,wave,volume) {
        if(!audio || audio.state==="closed") return;
        const osc=audio.createOscillator();
        const gain=audio.createGain();
        osc.type=wave;
        osc.frequency.setValueAtTime(freq,start);
        gain.gain.setValueAtTime(.0001,start);
        gain.gain.linearRampToValueAtTime(volume,start+.025);
        gain.gain.setValueAtTime(volume,start+duration*.45);
        gain.gain.exponentialRampToValueAtTime(.0001,start+duration);
        osc.connect(gain);
        gain.connect(audio.destination);
        osc.start(start);
        osc.stop(start+duration+.03);
      }

      function musicBeat() {
        if(!audio || audio.state==="closed") return;
        const t=audio.currentTime+.04;
        const n=melody[step%melody.length];

        note(n,t,.25,"triangle",.11);
        note(n*2,t+.02,.14,"sine",.025);

        if(step%3===0) {
          note(bass[Math.floor(step/3)%bass.length],t,.32,"sine",.075);
        }
        if(step%2===0) {
          note(melody[(step+5)%melody.length]*1.5,
               t+.09,.16,"sine",.022);
        }
        step++;
      }

      musicBeat();
      setInterval(musicBeat,240);
    } catch(err) {
      $("message").textContent="Music could not start in this browser.";
    }
  }

  reset();
  requestAnimationFrame(loop);
});
